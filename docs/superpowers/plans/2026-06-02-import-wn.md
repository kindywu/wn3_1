```python
# ── 解压模块 ──

def find_archives() -> List[Path]:
    """扫描 asset/ 目录，返回所有 wn*.dict.tar.gz 文件路径。"""
    if not ASSET_DIR.exists():
        print(f"[错误] asset 目录不存在: {ASSET_DIR}")
        sys.exit(1)
    archives = sorted(ASSET_DIR.glob("wn*.dict.tar.gz"))
    if not archives:
        print(f"[错误] 未在 {ASSET_DIR} 中找到 wn*.dict.tar.gz 文件")
        sys.exit(1)
    return archives


def parse_version(archive_path: Path) -> str:
    """从文件名提取版本号。如 wn3.1.dict.tar.gz → '3.1'"""
    name = archive_path.name
    m = re.search(r'wn(\d+\.\d+)\.', name)
    if not m:
        raise ValueError(f"无法从文件名提取版本号: {name}")
    return m.group(1)


def extract_archive(archive_path: Path) -> Path:
    """解压 tar.gz 到 dict/ 目录，返回解压后的 dict/ 路径。"""
    version = parse_version(archive_path)
    print(f"\n[解压] {archive_path.name} (版本 {version})")

    DICT_DIR.mkdir(parents=True, exist_ok=True)

    # 清空 dict/ 目录中的旧文件（确保干净解压）
    for f in DICT_DIR.iterdir():
        if f.is_file():
            f.unlink()
        elif f.is_dir():
            import shutil
            shutil.rmtree(f)

    with tarfile.open(archive_path, "r:gz") as tar:
        tar.extractall(path=DICT_DIR.parent, filter="data")

    # 验证关键文件存在
    required = ["data.noun", "data.verb", "data.adj", "data.adv",
                "index.noun", "index.verb", "index.adj", "index.adv",
                "index.sense"]
    for fname in required:
        if not (DICT_DIR / fname).exists():
            raise FileNotFoundError(f"解压后缺少关键文件: {fname}")

    print(f"  解压完成 → {DICT_DIR}")
    return DICT_DIR
```

```python
# ── 解析器模块 ──

def parse_data_file(filepath: Path, pos: str) -> Iterator[Tuple[Dict, List[Dict]]]:
    """解析 data.* 文件，逐行 yield (synset_row, [pointer_rows...])"""
    with open(filepath, "r", encoding="utf-8") as f:
        for line in f:
            line = line.strip()
            if not line or line.startswith("  ") or line.startswith("This"):
                continue

            head, _, gloss = line.partition(" | ")
            if not gloss:
                continue

            parts = head.split()
            offset = parts[0]
            lex_filenum = int(parts[1])
            ss_type = parts[2]
            w_cnt = int(parts[3], 16)

            idx = 4
            words = []
            for _ in range(w_cnt):
                word = parts[idx]
                lex_id = int(parts[idx + 1], 16)
                words.append((word, lex_id))
                idx += 2

            p_cnt = int(parts[idx])
            idx += 1
            pointers = []
            for _ in range(p_cnt):
                symbol = parts[idx]
                t_offset = parts[idx + 1]
                t_pos = parts[idx + 2]
                hex_str = parts[idx + 3]
                src_w = int(hex_str[:2], 16) if len(hex_str) >= 4 else 0
                tgt_w = int(hex_str[2:4], 16) if len(hex_str) >= 4 else 0
                pointers.append({
                    "symbol": symbol,
                    "target_synset_id": f"{t_offset}{t_pos}",
                    "source_w_num": src_w,
                    "target_w_num": tgt_w,
                })
                idx += 4

            # 动词 frame 解析
            frames = []
            if pos == "v":
                frame_match = re.findall(r'\+\s*(\d+)\s+(\d+)', head)
                for m in frame_match:
                    frames.append(int(m[0]))

            synset_row = {
                "synset_id": f"{offset}{pos}",
                "offset": offset,
                "pos": ss_type,
                "lex_filenum": lex_filenum,
                "ss_type": ss_type,
                "gloss": gloss.strip(),
                "word_count": w_cnt,
                "words": words,
                "frames": frames,
            }
            yield (synset_row, pointers)
```

```python
def parse_index_file(filepath: Path, pos: str) -> Iterator[Dict]:
    """解析 index.* 文件，逐行 yield lexical_entry row"""
    with open(filepath, "r", encoding="utf-8") as f:
        for line in f:
            line = line.strip()
            if not line or line.startswith("  ") or line.startswith("This"):
                continue
            parts = line.split()
            lemma = parts[0]
            p_idx = 2
            synset_cnt = int(parts[p_idx]); p_idx += 1
            ptr_cnt = int(parts[p_idx]); p_idx += 1
            ptr_symbols = " ".join(parts[p_idx:p_idx + ptr_cnt])
            p_idx += ptr_cnt
            sense_cnt = int(parts[p_idx]); p_idx += 1
            tagged_sense_cnt = int(parts[p_idx]); p_idx += 1
            offsets = parts[p_idx:p_idx + synset_cnt]

            yield {
                "lemma": lemma,
                "pos": pos,
                "synset_count": synset_cnt,
                "tagged_sense_count": tagged_sense_cnt,
                "pointer_symbols": ptr_symbols,
                "offsets": offsets,
```

```python
def parse_sense_index(filepath: Path) -> Iterator[Dict]:
    """解析 index.sense 文件，逐行 yield sense info"""
    with open(filepath, "r", encoding="utf-8") as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            parts = line.split()
            if len(parts) < 4:
                continue
            sense_key = parts[0]
            offset = parts[1]
            sense_number = int(parts[2])
            tag_cnt = int(parts[3])

            # 从 sense_key 解析 pos: lemma%ss_type:...
            m = re.search(r'%(\d+)', sense_key)
            if m:
                ss_type_num = int(m.group(1))
                pos = SS_TYPE_FROM_CODE.get(ss_type_num, "n")
            else:
                pos = "n"

            yield {
                "sense_key": sense_key,
                "offset": offset,
                "pos": pos,
                "sense_number": sense_number,
                "tag_cnt": tag_cnt,
            }
```

```python
def parse_exc(filepath: Path, pos: str) -> Iterator[Dict]:
    """解析 *.exc 文件，逐行 yield morph_exception row"""
    with open(filepath, "r", encoding="utf-8") as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            parts = line.split()
            if len(parts) >= 2:
                yield {
                    "pos": pos,
                    "surface_form": parts[0],
                    "base_form": parts[1],
                }


def parse_cntlist(filepath: Path) -> Iterator[Dict]:
    """解析 cntlist 文件，逐行 yield (sense_key, count, rank)"""
    with open(filepath, "r", encoding="utf-8") as f:
        for rank, line in enumerate(f, 1):
            parts = line.strip().split()
            if len(parts) < 3:
                continue
            yield {
                "sense_key": parts[1],
                "count": int(parts[0]),
                "rank": rank,
            }


def parse_frames(filepath: Path) -> Iterator[Tuple[int, str]]:
    """解析 verb.Framestext，yield (frame_number, frame_text)"""
    with open(filepath, "r", encoding="utf-8") as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            parts = line.split(None, 1)
            if len(parts) == 2:
                yield (int(parts[0]), parts[1].strip())


def parse_sents(filepath: Path) -> Iterator[Tuple[int, str]]:
    """解析 sents.vrb，yield (sentence_number, text)"""
    with open(filepath, "r", encoding="utf-8") as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            m = re.match(r'(\d+)\s+(.*)', line)
            if m:
                yield (int(m.group(1)), m.group(2))


def parse_sentidx(filepath: Path) -> Iterator[Tuple[str, List[int]]]:
    """解析 sentidx.vrb，yield (sense_key, [sentence_numbers])"""
    with open(filepath, "r", encoding="utf-8") as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            parts = line.split(None, 1)
            if len(parts) == 2:
                nums = [int(x.strip()) for x in parts[1].split(",")]
                yield (parts[0], nums)
```

```python
# ── 导入器模块 ──

def get_conn() -> sqlite3.Connection:
    """获取数据库连接并执行 DDL"""
    DB_DIR.mkdir(parents=True, exist_ok=True)
    conn = sqlite3.connect(str(DB_PATH))
    conn.executescript(CREATE_TABLES_SQL)
    conn.commit()
    return conn


def import_synsets_and_pointers(conn: sqlite3.Connection, dict_dir: Path, version: str) -> Tuple[int, int, Dict[str, int], Dict[str, list]]:
    """导入 data.* 文件 → synset + pointer 表。返回 (synset_count, pointer_count, offset_to_sense_keys, synset_frames_map)"""
    synset_count = 0
    pointer_count = 0
    # offset+pos → [sense_key, ...] (从 index.sense 预加载的映射)
    offset_to_sense_keys: Dict[str, int] = {}  # offset+pos → sense rowid (先暂存 offset 字符串)
    synset_frames: Dict[str, List[int]] = {}  # synset_id → [frame_numbers]

    cur = conn.cursor()

    synset_batch = []
    pointer_batch = []

    for pos in POS_FILES:
        data_path = dict_dir / f"data.{POS_TO_FILE[pos]}"
        if not data_path.exists():
            print(f"  [警告] 文件不存在: {data_path}")
            continue

        for synset_row, pointers in parse_data_file(data_path, pos):
            synset_id = synset_row["synset_id"]

            synset_batch.append((
                synset_id, version, synset_row["offset"], synset_row["pos"],
                synset_row["lex_filenum"], synset_row["ss_type"],
                synset_row["gloss"], synset_row["word_count"]
            ))

            # 暂存 words 用于后续 sense 匹配
            # 暂存 frames 用于后续 sense_frame 匹配
            if synset_row["frames"]:
                synset_frames[synset_id] = synset_row["frames"]

            for p in pointers:
                pointer_batch.append((
                    p["symbol"], synset_id, p["target_synset_id"],
                    p["source_w_num"], p["target_w_num"], version
                ))
                pointer_count += 1

            synset_count += 1

            if len(synset_batch) >= BATCH_SIZE:
                cur.executemany(
                    "INSERT OR IGNORE INTO synset(synset_id, version, offset, pos, lex_filenum, ss_type, gloss, word_count) VALUES(?,?,?,?,?,?,?,?)",
                    synset_batch
                )
                synset_batch.clear()

            if len(pointer_batch) >= BATCH_SIZE:
                cur.executemany(
                    "INSERT OR IGNORE INTO pointer(symbol, source_synset_id, target_synset_id, source_w_num, target_w_num, version) VALUES(?,?,?,?,?,?)",
                    pointer_batch
                )
                pointer_batch.clear()

    # 刷入剩余批次
    if synset_batch:
        cur.executemany(
            "INSERT OR IGNORE INTO synset(synset_id, version, offset, pos, lex_filenum, ss_type, gloss, word_count) VALUES(?,?,?,?,?,?,?,?)",
            synset_batch
        )
    if pointer_batch:
        cur.executemany(
            "INSERT OR IGNORE INTO pointer(symbol, source_synset_id, target_synset_id, source_w_num, target_w_num, version) VALUES(?,?,?,?,?,?)",
            pointer_batch
        )

    conn.commit()
    print(f"  synset: {synset_count} 行")
    print(f"  pointer: {pointer_count} 行 (暂缺 sense 引用，后续补全)")
    return synset_count, pointer_count, synset_frames


def import_lexical_entries(conn: sqlite3.Connection, dict_dir: Path, version: str) -> Tuple[int, Dict[Tuple[str, str], Tuple[int, List[str]]]]:
    """导入 index.* 文件 → lexical_entry 表。
    返回 (entry_count, {(lemma,pos): (entry_id, [offsets])})"""
    entry_count = 0
    lemma_offset_map: Dict[Tuple[str, str], Tuple[int, List[str]]] = {}

    cur = conn.cursor()
    batch = []

    for pos in POS_FILES:
        index_path = dict_dir / f"index.{POS_TO_FILE[pos]}"
        if not index_path.exists():
            print(f"  [警告] 文件不存在: {index_path}")
            continue

        for row in parse_index_file(index_path, pos):
            batch.append((
                row["lemma"], row["pos"], version,
                row["synset_count"], row["tagged_sense_count"],
                row["pointer_symbols"]
            ))

            if len(batch) >= BATCH_SIZE:
                cur.executemany(
                    "INSERT OR IGNORE INTO lexical_entry(lemma, pos, version, synset_count, tagged_sense_count, pointer_symbols) VALUES(?,?,?,?,?,?)",
                    batch
                )
                batch.clear()

    if batch:
        cur.executemany(
            "INSERT OR IGNORE INTO lexical_entry(lemma, pos, version, synset_count, tagged_sense_count, pointer_symbols) VALUES(?,?,?,?,?,?)",
            batch
        )

    conn.commit()

    # 构建 (lemma,pos) → (entry_id, offsets) 映射
    # 重新读取 index 文件获取 offsets
    for pos in POS_FILES:
        index_path = dict_dir / f"index.{POS_TO_FILE[pos]}"
        if not index_path.exists():
            continue
        for row in parse_index_file(index_path, pos):
            entry = cur.execute(
                "SELECT entry_id FROM lexical_entry WHERE lemma=? AND pos=? AND version=?",
                (row["lemma"], row["pos"], version)
            ).fetchone()
            if entry:
                lemma_offset_map[(row["lemma"], row["pos"])] = (entry[0], row["offsets"])
                entry_count += 1

    print(f"  lexical_entry: {entry_count} 行")
    return entry_count, lemma_offset_map
```

1. 从 `index.sense` 读取 sense_key → offset+pos+tag_cnt 映射
2. 从 `lemma_offset_map` 将 sense_key 的 lemma 部分匹配到 entry_id
3. 插入 sense 记录

```python
def import_senses(conn: sqlite3.Connection, dict_dir: Path, version: str,
                  synset_frames: Dict[str, List[int]],
                  lemma_offset_map: Dict[Tuple[str, str], Tuple[int, List[str]]]) -> Tuple[int, Dict[str, int]]:
    """导入 index.sense → sense 表。返回 (sense_count, {sense_key: sense_id})"""
    sense_count = 0
    sense_key_to_id: Dict[str, int] = {}

    cur = conn.cursor()
    batch = []

    # 先构建 sense_key → entry_id 的快速查找
    # sense_key 格式: lemma%...
    # lemma 是 sense_key 中 % 之前的部分
    # 通过 lemma+pos 匹配 lexical_entry

    # 先加载所有 lexical_entry 的 (lemma, pos) → entry_id 映射
    entry_lookup = {}
    for (lemma, pos), (entry_id, _) in lemma_offset_map.items():
        entry_lookup[(lemma, pos)] = entry_id

    sense_path = dict_dir / "index.sense"
    if not sense_path.exists():
        print("  [警告] index.sense 不存在，跳过 sense 导入")
        return 0, sense_key_to_id

    skipped = 0
    for row in parse_sense_index(sense_path):
        sense_key = row["sense_key"]
        offset = row["offset"]
        pos = row["pos"]
        synset_id = f"{offset}{pos}"

        # 从 sense_key 提取 lemma: 取 % 之前的部分
        sk_lemma = sense_key.split("%")[0].lower()

        # 匹配 entry_id
        entry_id = entry_lookup.get((sk_lemma, pos))
        if entry_id is None:
            skipped += 1
            continue

        # 从 sense_key 提取 lex_id
        # 格式: lemma%ss_type:lex_filenum:lex_id[:...]
        lex_id = 0
        m = re.search(r'%(\d+):(\d+):(\d+)', sense_key)
        if m:
            lex_id = int(m.group(3))

        batch.append((
            synset_id, entry_id, sense_key,
            row["sense_number"], row["tag_cnt"], lex_id, version
        ))

        sense_count += 1

        if len(batch) >= BATCH_SIZE:
            cur.executemany(
                "INSERT OR IGNORE INTO sense(synset_id, entry_id, sense_key, sense_number, tag_cnt, lex_id, version) VALUES(?,?,?,?,?,?,?)",
                batch
            )
            batch.clear()

    if batch:
        cur.executemany(
            "INSERT OR IGNORE INTO sense(synset_id, entry_id, sense_key, sense_number, tag_cnt, lex_id, version) VALUES(?,?,?,?,?,?,?)",
            batch
        )

    conn.commit()

    # 构建 sense_key → sense_id 映射
    rows = cur.execute("SELECT sense_id, sense_key FROM sense WHERE version=?", (version,)).fetchall()
    for sid, sk in rows:
        sense_key_to_id[sk] = sid

    if skipped > 0:
        print(f"  sense: {sense_count} 行 (跳过 {skipped} 条无匹配 entry 的记录)")
    else:
        print(f"  sense: {sense_count} 行")
    return sense_count, sense_key_to_id
```

```python
def resolve_pointer_senses(conn: sqlite3.Connection, dict_dir: Path, version: str):
    """补全 pointer 表中的 source_sense_id 和 target_sense_id。
    对于词级指针（source_w_num > 0 或 target_w_num > 0），
    需要找到对应的 sense 记录。"""
    cur = conn.cursor()

    # 构建 (synset_id, w_num) → sense_id 映射
    # w_num 对应 synset 中词的序号（1-based），对应 lex_id
    # 但 sense 表中的 lex_id 是词在 lex file 中的 ID，不是 w_num！
    # 需要从 data.* 重新解析 words 列表来建立映射
    #
    # 实际上，最简单的方法是：在 synset 中，词的序号 w_num (1-based)
    # 与 sense.lex_id 可能不一致。但绝大多数情况下 w_num-1 == lex_id。
    # 最准确的方式：重新解析 data.* 文件，拿到 (synset_id, word_index)
    # → sense_key 的对应关系。

    # 简化方案：直接用 w_num-1 作为 lex_id 去匹配
    # 因为 data.* 中的 w_num 实际上是词在 synset 中的序号（1-based），
    # 而 sense.lex_id 是词在 lex file 中的 ID。
    # 两者在绝大多数情况下一致（w_num-1 == lex_id），
    # 但为非精确匹配预留余量。

    # 为此，我们构建 (synset_id, lex_id) → sense_id 的索引
    cur.execute("""
        SELECT sense_id, synset_id, lex_id FROM sense WHERE version=?
    """, (version,))
    sense_index: Dict[Tuple[str, int], int] = {}
    for sid, synset_id, lex_id in cur.fetchall():
        sense_index[(synset_id, lex_id)] = sid

    # 更新所有词级 pointer
    # source_w_num > 0: 源是词级 → source_sense_id = sense_index[(source_synset, w_num-1)]
    # target_w_num > 0: 目标是词级 → target_sense_id = sense_index[(target_synset, w_num-1)]
    cur.execute("""
        SELECT pointer_id, source_synset_id, target_synset_id,
               source_w_num, target_w_num
        FROM pointer
        WHERE version=? AND (source_w_num > 0 OR target_w_num > 0)
    """, (version,))

    updates = []
    for pid, src_syn, tgt_syn, src_w, tgt_w in cur.fetchall():
        src_sense = None
        tgt_sense = None
        if src_w > 0:
            lex_id = src_w - 1  # w_num → lex_id (0-based)
            src_sense = sense_index.get((src_syn, lex_id))
        if tgt_w > 0:
            lex_id = tgt_w - 1
            tgt_sense = sense_index.get((tgt_syn, lex_id))
        if src_sense or tgt_sense:
            updates.append((src_sense, tgt_sense, pid))

    if updates:
        cur.executemany(
            "UPDATE pointer SET source_sense_id=?, target_sense_id=? WHERE pointer_id=?",
            updates
        )
        conn.commit()
        print(f"  pointer sense 引用补全: {len(updates)} 条词级指针")
```

```python
def count_file_lines(filepath: Path) -> int:
    """统计文件的非注释非空行数。"""
    count = 0
    with open(filepath, "r", encoding="utf-8") as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            if line.startswith("  ") or line.startswith("This"):
                continue
            count += 1
    return count


def validate(conn: sqlite3.Connection, dict_dir: Path, version: str,
             synset_count: int, pointer_count: int, sense_count: int, entry_count: int,
             aux_counts: Dict[str, int]) -> str:
    """执行四层校验，返回日志内容字符串。"""
    cur = conn.cursor()
    lines = []
    errors = []

    def log(msg: str):
        lines.append(msg)

    def err(msg: str):
        lines.append(f"  [FAIL] {msg}")
        errors.append(msg)

    def ok(msg: str):
        lines.append(f"  [PASS] {msg}")

    log(f"===== WordNet {version} 导入校验报告 =====")
    log(f"校验时间: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}")
    log(f"源文件: wn{version}.dict.tar.gz")
    log("")

    # ── Layer 1: 行数校验 ──
    log("--- 1. 行数校验 ---")

    checks = [
        ("data.noun", "SELECT COUNT(*) FROM synset WHERE pos='n' AND version=?", "n"),
        ("data.verb", "SELECT COUNT(*) FROM synset WHERE pos='v' AND version=?", "v"),
        ("data.adj", "SELECT COUNT(*) FROM synset WHERE pos IN ('a','s') AND version=?", "a"),
        ("data.adv", "SELECT COUNT(*) FROM synset WHERE pos='r' AND version=?", "r"),
        ("index.noun", "SELECT COUNT(*) FROM lexical_entry WHERE pos='n' AND version=?", "n"),
        ("index.verb", "SELECT COUNT(*) FROM lexical_entry WHERE pos='v' AND version=?", "v"),
        ("index.adj", "SELECT COUNT(*) FROM lexical_entry WHERE pos='a' AND version=?", "a"),
        ("index.adv", "SELECT COUNT(*) FROM lexical_entry WHERE pos='r' AND version=?", "r"),
        ("index.sense", "SELECT COUNT(*) FROM sense WHERE version=?", None),
    ]

    for fname, sql, pos_filter in checks:
        fpath = dict_dir / fname
        if not fpath.exists():
            log(f"  {fname}: 文件不存在，跳过")
            continue
        file_lines = count_file_lines(fpath)
        db_count = cur.execute(sql, (version,)).fetchone()[0]
        status = "PASS" if file_lines == db_count else "FAIL"
        log(f"  {fname}: 文件 {file_lines} 行 → 数据库 {db_count} 行 [{status}]")
        if status == "FAIL":
            err(f"行数不匹配: {fname} ({file_lines} vs {db_count})")

    # *.exc 合计校验
    exc_total = 0
    for pos in POS_FILES:
        exc_path = dict_dir / f"{POS_TO_FILE[pos]}.exc"
        if exc_path.exists():
            exc_total += count_file_lines(exc_path)
    cousin_path = dict_dir / "cousin.exc"
    if cousin_path.exists():
        exc_total += count_file_lines(cousin_path)
    db_exc = cur.execute("SELECT COUNT(*) FROM morph_exception WHERE version=?", (version,)).fetchone()[0]
    status = "PASS" if exc_total == db_exc else "FAIL"
    log(f"  *.exc (合计): 文件 {exc_total} 行 → 数据库 {db_exc} 行 [{status}]")
    if status == "FAIL":
        err(f"exc 行数不匹配: ({exc_total} vs {db_exc})")

    log("")

    # ── Layer 2: 引用完整性 ──
    log("--- 2. 引用完整性 ---")

    checks_fk = [
        ("pointer.source_synset → synset",
         "SELECT COUNT(*) FROM pointer WHERE version=? AND source_synset_id NOT IN (SELECT synset_id FROM synset)"),
        ("pointer.target_synset → synset",
         "SELECT COUNT(*) FROM pointer WHERE version=? AND target_synset_id NOT IN (SELECT synset_id FROM synset)"),
        ("sense.synset_id → synset",
         "SELECT COUNT(*) FROM sense WHERE version=? AND synset_id NOT IN (SELECT synset_id FROM synset)"),
        ("sense.entry_id → lexical_entry",
         "SELECT COUNT(*) FROM sense WHERE version=? AND entry_id NOT IN (SELECT entry_id FROM lexical_entry)"),
    ]

    for label, sql in checks_fk:
        orphan = cur.execute(sql, (version,)).fetchone()[0]
        if orphan == 0:
            ok(f"{label}: 0 孤立引用")
        else:
            err(f"{label}: {orphan} 条孤立引用")

    log("")

    # ── Layer 3: 全局统计 ──
    log("--- 3. 全局统计 ---")

    total_synset = cur.execute("SELECT COUNT(*) FROM synset WHERE version=?", (version,)).fetchone()[0]
    total_pointer = cur.execute("SELECT COUNT(*) FROM pointer WHERE version=?", (version,)).fetchone()[0]
    total_sense = cur.execute("SELECT COUNT(*) FROM sense WHERE version=?", (version,)).fetchone()[0]
    total_entry = cur.execute("SELECT COUNT(*) FROM lexical_entry WHERE version=?", (version,)).fetchone()[0]
    unique_lemma = cur.execute("SELECT COUNT(DISTINCT lemma) FROM lexical_entry WHERE version=?", (version,)).fetchone()[0]

    log(f"  Synset 总计:      {total_synset}")
    log(f"  Pointer 总计:     {total_pointer}")
    log(f"  Sense 总计:       {total_sense}")
    log(f"  LexicalEntry 总计: {total_entry}")
    log(f"  唯一词形:          {unique_lemma}")

    # WN 3.1 官方值校验
    if version == "3.1":
        known = {
            "synset": (total_synset, 117791),
            "pointer": (total_pointer, 225204),
            "sense": (total_sense, 207272),
        }
        for name, (actual, expected) in known.items():
            if actual == expected:
                ok(f"{name}: {actual} (官方值 {expected})")
            else:
                err(f"{name}: {actual} (官方值 {expected}, 差异 {abs(actual-expected)})")

    log("")

    if errors:
        log(f"===== 结果: {len(errors)} 项 FAIL =====")
        error_summary = "; ".join(errors)
    else:
        log("===== 结果: ALL PASS =====")
        error_summary = ""

    log_text = "\n".join(lines)
    return log_text, error_summary
```

```python
def main():
    print("=" * 60)
    print("  WordNet 字典 → SQLite 导入工具")
```
